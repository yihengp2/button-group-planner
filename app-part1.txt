  function isRover(c) { return c && c.pool === "rover"; }

  function groupSlots(group) {
    if (!Array.isArray(group.characterIds)) group.characterIds = [];
    while (group.characterIds.length < 3) group.characterIds.push(null);
    return group.characterIds.slice(0,3);
  }

  function usageCount(c, ignoreTarget = null) {
    if (!c) return 0;
    let count = 0;
    for (const g of state.groups) {
      const slots = groupSlots(g);
      for (let i=0;i<3;i++) {
        if (ignoreTarget && g.id === ignoreTarget.groupId && i === ignoreTarget.slotIndex) continue;
        const used = byId(slots[i]);
        if (!used) continue;
        if (isRover(c)) {
          if (isRover(used)) return 1;
        } else if (used.id === c.id) {
          count++;
        }
      }
    }
    return isRover(c) ? 0 : count;
  }

  function maxUses(c) { return isRover(c) ? 1 : Math.max(1, Number(c.maxUses)||1); }
  function remaining(c, ignoreTarget = null) {
    return Math.max(0, maxUses(c) - usageCount(c, ignoreTarget));
  }

  function totalCapacity() {
    let total = 1;
    for (const c of state.characters) if (!isRover(c)) total += maxUses(c);
    return total;
  }

  function totalUsed() {
    let total = 0;
    let rover = false;
    for (const g of state.groups) for (const id of groupSlots(g)) {
      const c = byId(id);
      if (!c) continue;
      if (isRover(c)) rover = true; else total++;
    }
    return total + (rover ? 1 : 0);
  }
